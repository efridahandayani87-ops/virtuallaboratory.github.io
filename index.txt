    <script>
        const DEV_PHOTO_BASE64 = "https://placehold.co/400x520/1c2541/ffffff?text=Efrida+Handayani+M";

        let audioCtx = null;
        let isMuted = false;

        function initAudio() {
            if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        }

        function playSound(type) {
            if (isMuted) return;
            try {
                initAudio();
                if (audioCtx.state === 'suspended') audioCtx.resume();
                const now = audioCtx.currentTime;

                if (type === 'click') {
                    const osc = audioCtx.createOscillator();
                    const gain = audioCtx.createGain();
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(600, now);
                    gain.gain.setValueAtTime(0.1, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.05);
                    osc.connect(gain); gain.connect(audioCtx.destination);
                    osc.start(now); osc.stop(now + 0.05);
                } else if (type === 'spark') {
                    const osc = audioCtx.createOscillator();
                    const gain = audioCtx.createGain();
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(800 + Math.random() * 400, now);
                    gain.gain.setValueAtTime(0.05, now);
                    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.04);
                    osc.connect(gain); gain.connect(audioCtx.destination);
                    osc.start(now); osc.stop(now + 0.04);
                }
            } catch(e) {}
        }

        function toggleAudio() {
            isMuted = !isMuted;
            const iconMb = document.getElementById('audio-icon-mb');
            const icon = document.getElementById('audio-icon');
            const cls = isMuted ? 'fa-solid fa-volume-xmark text-rose-400' : 'fa-solid fa-volume-high text-slate-300';
            if (iconMb) iconMb.className = cls;
            if (icon) icon.className = cls;
        }

        function toggleFullScreen() {
            if (!document.fullscreenElement) {
                document.documentElement.requestFullscreen().catch(() => {});
            } else {
                if (document.exitFullscreen) document.exitFullscreen();
            }
        }

        let simState = {
            running: false,
            time: 0,
            temp: 25,
            conc: 0.5,
            surfaceArea: 'kepingan',
            catalyst: false,
            volumeH2: 0,
            maxVol: 100,
            kRate: 0,
            graphPoints: [],
            lkpdLogs: []
        };

        function switchTab(tabId) {
            playSound('click');
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(el => el.classList.remove('active'));
            document.getElementById(`tab-${tabId}`).classList.remove('hidden');
            document.getElementById(`nav-${tabId}`).classList.add('active');
            if (tabId === 'simulasi') setTimeout(resizeAllCanvases, 100);
        }

        function onControlChange() {
            simState.temp = parseFloat(document.getElementById('slider-temp').value);
            simState.conc = parseFloat(document.getElementById('slider-conc').value);
            document.getElementById('val-temp').innerText = `${simState.temp} °C`;
            document.getElementById('val-conc').innerText = `${simState.conc.toFixed(1)} M`;
        }

        function setSurfaceArea(type) {
            playSound('click');
            simState.surfaceArea = type;
            document.getElementById('chip-kepingan').classList.toggle('active', type === 'kepingan');
            document.getElementById('chip-serbuk').classList.toggle('active', type === 'serbuk');
            reinitParticles();
        }

        function setCatalyst(val) {
            playSound('click');
            simState.catalyst = val;
            document.getElementById('chip-cat-off').classList.toggle('active', !val);
            document.getElementById('chip-cat-on').classList.toggle('active', val);
        }

        function toggleStartSim() {
            playSound('click');
            simState.running = !simState.running;
            const btn = document.getElementById('btn-start');
            document.getElementById('text-start').innerText = simState.running ? "Jeda Reaksi" : "Lanjutkan Reaksi";
            document.getElementById('icon-start').className = simState.running ? "fa-solid fa-pause" : "fa-solid fa-play";
        }

        function resetSimulation() {
            playSound('click');
            simState.running = false;
            simState.time = 0;
            simState.volumeH2 = 0;
            simState.graphPoints = [];
            document.getElementById('text-start').innerText = "Mulai Reaksi";

            document.getElementById('timer-badge').innerText = "t = 0.00s";

            drawGraph();
            drawMacro();
        }

        const microCanvas = document.getElementById('microCanvas');
        const ctxMicro = microCanvas.getContext('2d');
        let particles = [], mgGrid = [], sparks = [];

        function reinitParticles() {
            if (!microCanvas.width) return;
            particles = []; mgGrid = []; sparks = [];
            const w = microCanvas.width, h = microCanvas.height;

            const countH = Math.floor(15 + simState.conc * 20);
            for (let i = 0; i < countH; i++) {
                particles.push({
                    x: Math.random() * (w - 20) + 10,
                    y: Math.random() * (h - 60) + 10,
                    vx: (Math.random() - 0.5) * (1.2 + simState.temp / 30),
                    vy: (Math.random() - 0.5) * (1.2 + simState.temp / 30),
                    radius: 3, type: 'H+'
                });
            }

            const latticeY = h - 35;
            if (simState.surfaceArea === 'kepingan') {
                for (let x = w * 0.2; x < w * 0.8; x += 10) {
                    for (let y = latticeY; y < latticeY + 20; y += 10) {
                        mgGrid.push({ x: x, y: y, active: true });
                    }
                }
            } else {
                for (let i = 0; i < 4; i++) {
                    const startX = 20 + i * (w / 4);
                    for (let x = startX; x < startX + 25; x += 8) {
                        for (let y = latticeY; y < latticeY + 20; y += 8) {
                            mgGrid.push({ x: x, y: y, active: true });
                        }
                    }
                }
            }
        }

        function updateMicroPhysics() {
            const w = microCanvas.width, h = microCanvas.height;
            particles.forEach(p => {
                p.x += p.vx * (simState.running ? 1 : 0.2);
                p.y += p.vy * (simState.running ? 1 : 0.2);
                if (p.x < 5 || p.x > w - 5) p.vx *= -1;
                if (p.y < 5 || p.y > h - 40) p.vy *= -1;
            });

            if (simState.running) {
                let effProb = 0.03 * (simState.temp / 30) * simState.conc;
                if (simState.catalyst) effProb *= 2;

                particles.forEach(p => {
                    if (p.type === 'H+') {
                        mgGrid.forEach(mg => {
                            if (mg.active && Math.hypot(p.x - mg.x, p.y - mg.y) < 10) {
                                p.vy *= -1;
                                if (Math.random() < effProb) {
                                    p.type = 'H2'; mg.active = false;
                                    playSound('spark');
                                }
                            }
                        });
                    }
                });
            }
        }

        function drawMicro() {
            ctxMicro.clearRect(0, 0, microCanvas.width, microCanvas.height);
            mgGrid.forEach(mg => {
                if (mg.active) {
                    ctxMicro.fillStyle = '#f59e0b';
                    ctxMicro.fillRect(mg.x - 3, mg.y - 3, 6, 6);
                }
            });
            particles.forEach(p => {
                ctxMicro.beginPath();
                ctxMicro.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
                ctxMicro.fillStyle = p.type === 'H+' ? '#22d3ee' : '#34d399';
                ctxMicro.fill();
            });
        }

        const macroCanvas = document.getElementById('macroCanvas');
        const ctxMacro = macroCanvas.getContext('2d');

        function drawMacro() {
            ctxMacro.clearRect(0, 0, macroCanvas.width, macroCanvas.height);
            const w = macroCanvas.width, h = macroCanvas.height;

            ctxMacro.fillStyle = 'rgba(34, 211, 238, 0.2)';
            ctxMacro.fillRect(10, h - 35, w * 0.4, 25);
            ctxMacro.strokeStyle = '#64748b';
            ctxMacro.strokeRect(10, 10, w * 0.4, h - 20);

            const fillW = (w * 0.4) * Math.min(simState.volumeH2 / 100, 1);
            ctxMacro.fillStyle = 'rgba(52, 211, 153, 0.4)';
            ctxMacro.fillRect(w * 0.55, 15, fillW, 20);
            ctxMacro.strokeRect(w * 0.55, 15, w * 0.4, 20);

            ctxMacro.fillStyle = '#34d399';
            ctxMacro.font = '10px monospace';
            ctxMacro.fillText(`Gas H2: ${simState.volumeH2.toFixed(1)} mL`, w * 0.55, 10);
        }

        const graphCanvas = document.getElementById('graphCanvas');
        const ctxGraph = graphCanvas.getContext('2d');

        function drawGraph() {
            ctxGraph.clearRect(0, 0, graphCanvas.width, graphCanvas.height);
            const w = graphCanvas.width, h = graphCanvas.height;
            
            ctxGraph.strokeStyle = '#3a506b';
            ctxGraph.beginPath(); ctxGraph.moveTo(25, 5); ctxGraph.lineTo(25, h - 15); ctxGraph.lineTo(w - 5, h - 15); ctxGraph.stroke();

            if (simState.graphPoints.length > 1) {
                ctxGraph.strokeStyle = '#4ea8de';
                ctxGraph.lineWidth = 2;
                ctxGraph.beginPath();
                const maxT = Math.max(30, simState.time);
                simState.graphPoints.forEach((pt, i) => {
                    const x = 25 + (pt.t / maxT) * (w - 35);
                    const y = (h - 15) - (pt.v / 100) * (h - 25);
                    if (i === 0) ctxGraph.moveTo(x, y); else ctxGraph.lineTo(x, y);
                });
                ctxGraph.stroke();
            }
        }

        function resizeAllCanvases() {
            [microCanvas, macroCanvas, graphCanvas].forEach(c => {
                if (c && c.parentElement) {
                    c.width = c.parentElement.clientWidth;
                    c.height = c.parentElement.clientHeight;
                }
            });
            reinitParticles();
            drawGraph();
            drawMacro();
        }
        window.addEventListener('resize', resizeAllCanvases);

        let lastTime = performance.now();
        function mainLoop(now) {
            const dt = (now - lastTime) / 1000;
            lastTime = now;

            if (simState.running) {
                simState.kRate = 0.02 * Math.pow(1.5, (simState.temp - 25) / 10) * simState.conc * (simState.surfaceArea === 'serbuk' ? 2 : 1) * (simState.catalyst ? 2 : 1);
                simState.time += dt * 1.5;
                simState.volumeH2 = simState.maxVol * (1 - Math.exp(-simState.kRate * simState.time));
                simState.graphPoints.push({ t: simState.time, v: simState.volumeH2 });

                document.getElementById('timer-badge').innerText = `t = ${simState.time.toFixed(2)}s`;

                if (simState.volumeH2 >= 99.5) {
                    simState.volumeH2 = 100;
                    simState.running = false;
                }
            }

            updateMicroPhysics();
            drawMicro();
            drawMacro();
            drawGraph();
            requestAnimationFrame(mainLoop);
        }

            if (simState.time === 0) return alert("Jalankan simulasi dulu!");
